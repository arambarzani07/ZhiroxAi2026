  1939	func isManagedVersionArtifact(name string) bool {
  1940		n := strings.ToLower(strings.TrimSpace(name))
  1941		// Only files created by this updater are eligible.
  1942		if n == "zhiroxaisupervisor.exe.bak" ||
  1943			n == "zhiroxaisupervisor.exe.new" ||
  1944			n == "zhiroxaisupervisor.exe.old" {
  1945			return true
  1946		}
  1947		if strings.HasPrefix(n, "zhiroxaisupervisor-") &&
  1948			(strings.HasSuffix(n, ".bak") || strings.HasSuffix(n, ".old") || strings.HasSuffix(n, ".new")) {
  1949			return true
  1950		}
  1951		if strings.HasPrefix(n, "update-") &&
  1952			(strings.HasSuffix(n, ".tmp") || strings.HasSuffix(n, ".part")) {
  1953			return true
  1954		}
  1955		return false
  1956	}
